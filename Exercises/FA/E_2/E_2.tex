	\section{سوال اول}
	
	فرض کنید سه بردار ثابت $\mathbf{a}, \mathbf{b}, \mathbf{c}$ از مبدأ (اصل مختصات) به نقاط $A, B, C$ کشیده شده‌اند.
	
	\begin{enumerate}
		\item فاصله مبدأ تا صفحه‌ای که از نقاط $A, B, C$ می‌گذرد چقدر است؟
		\item مساحت مثلث $ABC$ چقدر است؟
	\end{enumerate}

	\section{سوال دوم}
	
	
	نشان دهید که:
	
	\begin{enumerate}
		\item $\nabla r^n = n\, r^{\,n-2} \,\vec{r}$
		\item $\nabla^2 (\ln r) = \dfrac{1}{r^2}$
	\end{enumerate}
	
	\textbf{توجه:} برای آشنایی با عملگر لاپلاسین و نحوه تعریف آن در دستگاه‌های مختصات مختلف، پیوست را مطالعه کنید.
	
	
	\section{سوال سوم}
	مؤلفه‌های بردار شتاب $\mathbf{a}$ را در مختصات استوانه پیدا کنید.
	
	\[
	\mathbf{a} = a_r \,\hat{r} + a_\theta \,\hat{\theta} + a_z \,\hat{z}
	\]
	
	\section{سوال چهارم}
	
	ماتریس تبدیل (تغییر مختصات) را پیدا کنید که یک دستگاه مختصات را به اندازه $120^\circ$ حول محوری بچرخاند که با محورهای $x$ و $y$ زاویه برابر و با محور $z$ زاویه $30^\circ$ می‌سازد.
	
	\section{سوال پنجم}
	
	
	یک ذره با سرعت ثابت $v = {const}$ بر روی منحنی 
	\[
	r = k(1 + \cos \theta) \quad (\text{کاردیوئید})
	\]
	حرکت می‌کند.
	
	\begin{enumerate}
		\item مؤلفه شتاب در راستای شعاعی: $\ddot{\mathbf{r}} \cdot \hat{\mathbf{e}}_r = \mathbf{a} \cdot \hat{\mathbf{e}}_r$
		\item اندازه شتاب: $|\mathbf{a}|$
		\item مشتق زاویه‌ای: $\dot{\theta}$
	\end{enumerate}
	را محاسبه کنید.
	
	\section{سؤال ششم}
	
	مقدار انتگرال سطحی زیر را پیدا کنید:
	\[
	\int_S \mathbf{A} \cdot d\mathbf{a}
	\]
	که:
	\[
	\mathbf{A} = (x^2 + y^2 + z^2)(x \hat{\mathbf{i}} + y \hat{\mathbf{j}} + z \hat{\mathbf{k}})
	\]
	و $S$ سطح بسته‌ای است که توسط کره
	\[
	x^2 + y^2 + z^2 = R^2
	\]
	تعریف شده است.
	
	\begin{enumerate}
		\item انتگرال را به صورت مستقیم محاسبه کنید.
		\item انتگرال را با استفاده از قضیه گاوس  محاسبه کنید:
		\[
		\int_S \mathbf{A} \cdot d\mathbf{a} = \int_V (\nabla \cdot \mathbf{A}) \, dV
		\]
	\end{enumerate}

	\section{سوال هفتم}
	
	با بسط مستقیم نشان دهید که:
	\[
	|\Lambda|^2 = 1
	\]
	برای سادگی، فرض کنید \(\Lambda\) یک ماتریس تبدیل متعامد دوبعدی است.
	
	\section{سوال امتیازی}
	با استفاده از تعریف لاپلاسین به‌صورت:
	\[
	\nabla^2 f = \nabla \cdot (\nabla f)
	\]
	و روابط گرادیان و دیورژانس در مختصات استوانه‌ای، 
	رابطه‌ی صریح لاپلاسین را در دستگاه مختصات استوانه‌ای \((r, \phi, z)\) به‌دست آورید.
	\[
	\nabla^2 f =
	\frac{1}{r} \frac{\partial}{\partial r}
	\left( r \frac{\partial f}{\partial r} \right)
	+ \frac{1}{r^2} \frac{\partial^2 f}{\partial \theta^2}
	+ \frac{\partial^2 f}{\partial z^2}
	\]

	\section{پیوست}
	
	\hrule\vspace{1em}
	\subsection*{تعریف لاپلاسین در دستگاه‌های مختصات مختلف}
	
	
	\[
	\nabla^2 f = \nabla \cdot (\nabla f)
	\]
	
	عملگر لاپلاسین برای یک تابع اسکالر $f$ در دستگاه‌های مختصات مختلف به‌صورت زیر تعریف می‌شود:
	
	
	
	\subsection*{۱. مختصات دکارتی $(x, y, z)$}
	\[
	\nabla^2 f =
	\frac{\partial^2 f}{\partial x^2} +
	\frac{\partial^2 f}{\partial y^2} +
	\frac{\partial^2 f}{\partial z^2}
	\]
	
	\subsection*{۲. مختصات استوانه‌ای $(r, \theta, z)$}
	\[
	\nabla^2 f =
	\frac{1}{r} \frac{\partial}{\partial r}
	\left( r \frac{\partial f}{\partial r} \right)
	+ \frac{1}{r^2} \frac{\partial^2 f}{\partial \theta^2}
	+ \frac{\partial^2 f}{\partial z^2}
	\]
	
	\subsection*{۳. مختصات کروی $(r, \theta, \phi)$}
	\[
	\nabla^2 f =
	\frac{1}{r^2} \frac{\partial}{\partial r}
	\left( r^2 \frac{\partial f}{\partial r} \right)
	+ \frac{1}{r^2 \sin\theta} \frac{\partial}{\partial \theta}
	\left( \sin\theta \frac{\partial f}{\partial \theta} \right)
	+ \frac{1}{r^2 \sin^2\theta}
	\frac{\partial^2 f}{\partial \phi^2}
	\]
